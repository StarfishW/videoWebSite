Python:

import requests

    r = requests.post("{{ base }}/api/videos",
                      headers={"Authorization": "Bearer {{ token }}"},
                      files={"file": f},
                      data={"title": "Lecture 1", "category": "Mathematics"})
print(r.status_code, r.json())
