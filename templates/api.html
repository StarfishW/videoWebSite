API = "<API URL>"
with open("lecture01.mp4", "rb") as f:
    r = requests.post(API + "/videos",
                      headers={"Authorization": "Bearer <token>"},
                      files={"file": f},
                      data={"title": "Lecture 1", "category": "Mathematics"})
print(r.status_code, r.json())
